#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#define X1 -10.0
#define X2 10.0
#define N 1024
#define h ((X2-X1)/(N-1))
float fx(float x) {
    return 8*x/(x*x+1);
}
int main() {
    float *X, *FX;
    float *DX, *DFX;
    int i;
    X  = (float*) malloc(N * sizeof(float));
    FX = (float*) malloc(N * sizeof(float));
    for (i = 0; i < N; i++) {
        X[i]  = X1 + i*h;
        FX[i] = fx(X[i]);
    }
    /* Code A: Allocate memory on GPU */
    cudaMalloc((void**)&DX,  N * sizeof(float));
    cudaMalloc((void**)&DFX, N * sizeof(float));
    /* Code B: Copy input from CPU to GPU */
    cudaMemcpy(DX, FX, N * sizeof(float), cudaMemcpyHostToDevice);
    /* Define grid and block */
    dim3 blocksize(256);
    dim3 gridsize((N + blocksize.x - 1) / blocksize.x);
    /* Code C: Invoke kernel */
    CentralDifference<<<gridsize, blocksize>>>(DFX, DX);
    /* Code D: Copy output from GPU to CPU */
    cudaMemcpy(FX, DFX, N * sizeof(float), cudaMemcpyDeviceToHost);
    /* Code E: Free memory on CPU and GPU */
    free(X);
    free(FX);
    cudaFree(DX);
    cudaFree(DFX);
    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <mpi.h>
#define M 5
int main(int argc, char *argv[]) {
    int i, rank, size;
    int *A, *Ac;
    MPI_Status status;
    MPI_Init(&argc, &argv);
    MPI_Comm_size(MPI_COMM_WORLD, &size);
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    Ac = (int*) malloc(M * sizeof(int));
    for (i = 0; i < M; i++)
        Ac[i] = i;
    /* Code E: All processes send its Ac to process 1 */
    MPI_Send(Ac, M, MPI_INT, 1, 0, MPI_COMM_WORLD);
/* Code F: Process 1 receive and print A */
    if (rank == 1) printf("%d\n", (MPI_Recv(Ac, M, MPI_INT, MPI_ANY_SOURCE, 0, MPI_COMM_WORLD, &status), Ac[0]));
    MPI_Finalize();
    return 0;
}
